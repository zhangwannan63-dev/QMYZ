cmake_minimum_required(VERSION 3.18)
project(qmyz_cpp VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(qmyz_core src/core.cpp src/http.cpp src/client.cpp)
target_include_directories(qmyz_core PUBLIC include ${JSON_INCLUDE_DIR})
target_link_libraries(qmyz_core PUBLIC CURL::libcurl OpenSSL::Crypto)
target_compile_options(qmyz_core PRIVATE -Wall -Wextra -Wpedantic)
foreach(app qmyz qm_cli qm_auto fight)
  add_executable(${app} src/main.cpp)
  target_link_libraries(${app} PRIVATE qmyz_core)
  target_compile_definitions(${app} PRIVATE QMYZ_ENTRY="${app}")
endforeach()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE qmyz_core)
add_test(NAME core COMMAND core_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME mock_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:qmyz>)
endif()
install(TARGETS qmyz qm_cli qm_auto fight RUNTIME DESTINATION bin)
